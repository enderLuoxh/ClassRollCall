cmake_minimum_required(VERSION 3.24)
project(ClassRollCall VERSION 2.0.0 LANGUAGES C CXX RC)
if(NOT WIN32)
  message(FATAL_ERROR "ClassRollCall requires Windows")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
if(MSVC)
  add_compile_options(/utf-8 /W4 /EHsc)
endif()
add_library(rollcall_core STATIC src/core/random.cpp src/platform/platform.cpp src/storage/store.cpp)
target_include_directories(rollcall_core PUBLIC src)
target_link_libraries(rollcall_core PUBLIC advapi32 shell32 user32 gdi32 ole32)
add_subdirectory(third_party)
add_library(rollcall_import STATIC src/importer/importer.cpp src/importer/text_csv.cpp src/importer/office.cpp src/importer/xls.cpp src/importer/xlsx.cpp src/importer/docx.cpp)
target_include_directories(rollcall_import PUBLIC src)
target_link_libraries(rollcall_import PUBLIC rollcall_core PRIVATE miniz pugixml::pugixml xlsreader)
add_executable(ClassRollCall WIN32 src/main.cpp src/ui/application.cpp src/ui/paint.cpp src/ui/dialogs.cpp src/resources/app.rc)
target_include_directories(ClassRollCall PRIVATE src/resources)
target_link_libraries(ClassRollCall PRIVATE rollcall_core rollcall_import gdiplus comctl32 comdlg32)
if(MSVC)
  target_link_options(ClassRollCall PRIVATE /SUBSYSTEM:WINDOWS,6.01 /MANIFEST:NO)
endif()
set_target_properties(ClassRollCall PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_SOURCE_DIR}/dist" RUNTIME_OUTPUT_DIRECTORY_RELEASE "${CMAKE_SOURCE_DIR}/dist")
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE rollcall_core)
  add_test(NAME core COMMAND core_tests)
  add_executable(import_tests tests/import_tests.cpp)
  target_link_libraries(import_tests PRIVATE rollcall_import)
  add_test(NAME import COMMAND import_tests "${CMAKE_SOURCE_DIR}/tests/fixtures/generated")
endif()
